Triton Flash Attention Kernel for Decoupled Attention: The Forward Pass

In the previous post, we introduced a decoupled parameterization of self-attention that separates self- and cross-token interactions through independent parameter sets. In this post, we explore an efficient FlashAttention implementation of this parameterization. The key idea is to tile the newly introduced self-interaction parameters in the same way as the query tokens, enabling self-interaction scores to be computed alongside the standard query-key interactions within each query block.

Decoupled Attention in Triton: The Forward Pass

import torch
import triton
import triton.language as tl
    
    
@triton.jit
def decoupled_attn_fwd_kernel(
    # --------------------------------------------------
    # Input / Output tensors
    # --------------------------------------------------
    q,         # Query: [B, T, HQ, DIM_QK]
    k,         # Key: [B, T, H, DIM_QK]
    a,         # Self-interaction: [B, T, HQ, DIM_A]
    v,         # Value: [B, T, H, DIM_V]
    o,         # Output: [B, T, HQ, DIM_V]

    # ----------------------------------------------------------
    # Attention parameters
    # ----------------------------------------------------------
    scale_qk,  # Scaling factor for cross-token interaction
    scale_a,   # Scaling factor for self-token interaction

    # ------------------------------------------------------------
    # Model dimensions
    # ------------------------------------------------------------
    T,                      # Sequence length (runtime parameter)
    B: tl.constexpr,        # Batch size (compile-time constant)
    HQ: tl.constexpr,       # Number of Query heads
    H: tl.constexpr,        # Number of KV heads
    G: tl.constexpr,        # GQA group size = HQ // H
    DIM_QK: tl.constexpr,   # Query / Key dimension (per head)
    DIM_A: tl.constexpr,    # Self-interaction dimension
    DIM_V: tl.constexpr,    # Value dimension

    # ------------------------------------------------------------------------
    # Triton tile dimensions
    # - Tiling is applied along q/k/a/v tokens, and value features
    # - No feature tiling is applied for q/k/a
    # ------------------------------------------------------------------------
    BT: tl.constexpr, # Query token block size
    BS: tl.constexpr, # Key / Value token block size
    BV: tl.constexpr, # Value feature block size
):
    # ----------------------------------------------------------------------
    # Index Arithmetic (Program ID mapping): tile the output into [BT, BV]
    # ----------------------------------------------------------------------

    # Number of Value feature blocks along the DIM_V dimension
    num_v_blocks = tl.cdiv(DIM_V, BV)

    # Decode program_id(0) into Query block index and Value block index
    # Query-block-major order: value block index changes faster
    pid_0 = tl.program_id(0).to(tl.int64)
    idx_t = pid_0 // num_v_blocks
    idx_v = pid_0 % num_v_blocks

    # Decode program_id(1) into Batch index and Query head index
    # Batch-major order: query head index changes faster
    pid_1 = tl.program_id(1).to(tl.int64)
    idx_b = pid_1 // HQ
    idx_hq = pid_1 % HQ

    # Map Query head to its corresponding KV head (GQA)
    # Each KV head is shared by G Query heads
    idx_h = idx_hq // G

    # Reciprocal of ln(2), used to implement exp(x) via exp2(x * RCP_LN2)
    RCP_LN2: tl.constexpr = 1.4426950216

    # Query token offsets within the current block
    offsets_q = idx_t * BT + tl.arange(0, BT)
    # Query feature dimension offsets (no feature tiling)
    offsets_d = tl.arange(0, DIM_QK)
    # Query pointers: [B, T, HQ, DIM_QK] -> [BT, DIM_QK]
    q_ptrs = (
        q
        + idx_b * T * HQ * DIM_QK
        + offsets_q[:, None] * HQ * DIM_QK
        + idx_hq * DIM_QK
        + offsets_d[None, :]
    )
    # Load Query tile in shape [BT, DIM_QK]
    q_mask = (offsets_q[:, None] < T) & (offsets_d[None, :] < DIM_QK)
    q_tile = tl.load(q_ptrs, mask=q_mask, other=0.0)

    # Self-interaction feature dimension offsets (no feature tiling)
    offsets_a = tl.arange(0, DIM_A)
    # Self-interaction pointers: [B, T, HQ, DIM_A] -> [BT, DIM_A]
    # Note that the tiling of a (token-level) is aligned with that of q
    a_ptrs = (
        a
        + idx_b * T * HQ * DIM_A
        + offsets_q[:, None] * HQ * DIM_A
        + idx_hq * DIM_A
        + offsets_a[None, :]
    )

    # Load Self-interaction tile in shape [BT, DIM_A]
    a_mask = (offsets_q[:, None] < T) & (offsets_a[None, :] < DIM_A)
    a_tile = tl.load(a_ptrs, mask=a_mask, other=0.0)
    # Compute self-interaction logits for the tile in shape [BT]
    self_logits = tl.sum(a_tile * a_tile, axis=1) * scale_a * RCP_LN2

    # Value feature offsets within the current block
    offsets_v = idx_v * BV + tl.arange(0, BV)

    # Output tile accumulator: [BT, BV]
    o_tile = tl.zeros([BT, BV], dtype=tl.float32)
    # Output pointers: [B, T, HQ, DIM_V] -> [BT, BV]
    o_ptrs = (
        o
        + idx_b * T * HQ * DIM_V
        + offsets_q[:, None] * HQ * DIM_V
        + idx_hq * DIM_V
        + offsets_v[None, :]
    )

    # Initialize Online Softmax States
    # Running maximum of attention logits for each query row
    running_max = tl.full(shape=[BT], value=float('-inf'), dtype=tl.float32)
    # Running softmax denominator for each query row
    running_denominator = tl.zeros(shape=[BT], dtype=tl.float32)

    # Iterate through the key blocks that are before the query block
    for idx_k in range(0, idx_t * BT, BS):
        offsets_k = idx_k + tl.arange(0, BS)
        # Key pointers: [B, T, H, DIM_QK] -> [DIM_QK, BS]
        k_ptrs = (
            k
            + idx_b * T * H * DIM_QK
            + offsets_k[None, :] * H * DIM_QK
            + idx_h * DIM_QK
            + offsets_d[:, None]
        )
        # Load Key tile in shape [DIM_QK, BS]
        # Note that < idx_t * BT implies < T by idx_t = pid_0 // num_v_blocks
        k_mask = (offsets_d[:, None] < DIM_QK) & (offsets_k[None, :] < idx_t * BT)
        k_tile = tl.load(k_ptrs, mask=k_mask, other=0.0)

        # Value pointers: [B, T, H, DIM_V] -> [BS, BV]
        v_ptrs = (
            v
            + idx_b * T * H * DIM_V
            + offsets_k[:, None] * H * DIM_V
            + idx_h * DIM_V
            + offsets_v[None, :]
        )
        # Load Value tile in shape [BS, BV]
        v_mask = (offsets_k[:, None] < idx_t * BT) & (offsets_v[None, :] < DIM_V)
        v_tile = tl.load(v_ptrs, mask=v_mask, other=0.0)

        # Compute qk logits score for the current Q-K tile in shape [BT, BS]
        qk_logits = tl.dot(q_tile, k_tile, input_precision="ieee") * scale_qk * RCP_LN2
        # Mask out invalid key positions to prevent them from contributing to softmax
        qk_logits = tl.where(offsets_k[None, :] < idx_t * BT, qk_logits, float('-inf'))

        # Online softmax: update the running maximum for every query row
        prev_max = running_max
        running_max = tl.maximum(prev_max, tl.max(qk_logits, axis=1))
        # Replace -inf with a finite value to avoid -inf - (-inf) = NaN
        safe_max = tl.where(running_max == float('-inf'), 0.0, running_max)
        # Fetch the new maximum as the rescaling factor
        rescale = tl.exp2(prev_max - safe_max)

        # Compute unnormalized softmax weights for the current Q-K tile
        softmax_numerators = tl.exp2(qk_logits - safe_max[:, None])
        # Compute unnormalized running O_tile
        o_tile = o_tile * rescale[:, None] + tl.dot(softmax_numerators.to(v_tile.dtype), v_tile, input_precision="ieee")

        # Update the accumulated softmax denominator for each query row for later use
        running_denominator = running_denominator * rescale + tl.sum(softmax_numerators, axis=1)

    # Iterate over key blocks from the start of the query block to the end
    for idx_k in range(idx_t * BT, tl.minimum(idx_t * BT + BT, T), BS):
        offsets_k = idx_k + tl.arange(0, BS)
        # Key pointers: [B, T, H, DIM_QK] -> [DIM_QK, BS]
        k_ptrs = (
            k
            + idx_b * T * H * DIM_QK
            + offsets_k[None, :] * H * DIM_QK
            + idx_h * DIM_QK
            + offsets_d[:, None]
        )
        # Load Key tile in shape [DIM_QK, BS]
        k_mask = (offsets_d[:, None] < DIM_QK) & (offsets_k[None, :] < tl.minimum(idx_t * BT + BT, T))
        k_tile = tl.load(k_ptrs, mask=k_mask, other=0.0)

        # Compute qk logits score for the current Q-K tile in shape [BT, BS]
        qk_logits = tl.dot(q_tile, k_tile, input_precision="ieee") * scale_qk * RCP_LN2
        # Replace diagonal (q == k) with self-interaction logits
        attn_logits = tl.where(offsets_q[:, None] == offsets_k[None, :], self_logits[:, None], qk_logits)

        # Apply causal masking and exclude out-of-bound Q/K positions
        causal_mask = offsets_q[:, None] >= offsets_k[None, :]
        valid_mask = (offsets_q[:, None] < T) & (offsets_k[None, :] < T)
        causal_logits = tl.where(causal_mask & valid_mask, attn_logits, float('-inf'))

        # Value pointers: [B, T, H, DIM_V] -> [BS, BV]
        v_ptrs = (
            v
            + idx_b * T * H * DIM_V
            + offsets_k[:, None] * H * DIM_V
            + idx_h * DIM_V
            + offsets_v[None, :]
        )
        # Load Value tile in shape [BS, BV]
        v_mask = (offsets_k[:, None] < tl.minimum(idx_t * BT + BT, T)) & (offsets_v[None, :] < DIM_V)
        v_tile = tl.load(v_ptrs, mask=v_mask, other=0.0)

        # Online softmax: update the running maximum for every query row
        prev_max = running_max
        running_max = tl.maximum(prev_max, tl.max(causal_logits, axis=1))
        # Replace -inf with a finite value to avoid -inf - (-inf) = NaN
        safe_max = tl.where(running_max == float('-inf'), 0.0, running_max)
        # Fetch the new maximum as the rescaling factor
        rescale = tl.exp2(prev_max - safe_max)

        # Compute unnormalized softmax weights for the current Q-K tile
        softmax_numerators = tl.exp2(causal_logits - safe_max[:, None])
        # Compute unnormalized running O_tile
        o_tile = o_tile * rescale[:, None] + tl.dot(softmax_numerators.to(v_tile.dtype), v_tile, input_precision="ieee")

        # Update the accumulated softmax denominator for each query row for later use
        running_denominator = running_denominator * rescale + tl.sum(softmax_numerators, axis=1)

    # Normalize the output using a safe denominator
    safe_denominator = tl.where(running_denominator > 0.0, running_denominator, 1.0)
    o_tile = o_tile / safe_denominator[:, None]

    # Mask out-of-bound query positions and value dimensions
    o_mask = (offsets_q[:, None] < T) & (offsets_v[None, :] < DIM_V)
    # Store the output, whose dtype is obtained from the pointer via o_ptrs.dtype.element_ty
    tl.store(o_ptrs, o_tile.to(o_ptrs.dtype.element_ty), mask=o_mask)
